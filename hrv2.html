Cài đặt Kết Nối Google Sheets (Tự Động 100%)

Đoạn mã v4.1 - Khai thác triệt để Apps Script (Đã tối ưu CORS):

✅ Tự nhận diện cột/tiêu đề động cho 4 sheet · ✅ Định dạng SĐT dạng text (không mất số 0) · ✅ Đóng băng dòng tiêu đề + tô màu Trạng thái tự động

✅ Tự động sao lưu (giữ 10 bản gần nhất trong sheet "_Backup") · ✅ Ghi nhật ký mọi lần đồng bộ vào sheet "_NhatKy"

✅ Gửi email tự động khi có Ứng viên mới chuyển trạng thái "Đã nhận" · ✅ Hỗ trợ tải riêng từng tab (?tab=ungVien) cho tốc độ nhanh hơn

// ==== HR IDOL RECRUIT - APPS SCRIPT BACKEND v4.1 (Bản đầy đủ, dán 1 lần) ====
const TABS = ['ungVien', 'tienIch', 'noiDung', 'luuTruKhac'];
const NOTIFY_EMAIL = ""; // Điền email HR vào đây để nhận thông báo mặc định
const MAX_BACKUPS = 10;

function doPost(e) {
  try {
    const doc = SpreadsheetApp.getActiveSpreadsheet();
    let data;

    if (e.parameter && e.parameter.data) {
      data = JSON.parse(e.parameter.data);
    } else if (e.postData && e.postData.contents) {
      data = JSON.parse(e.postData.contents);
    } else {
      throw new Error("Không nhận được dữ liệu hợp lệ.");
    }

    const before = readAllTabs(doc);
    const notifyEmail = (data._meta && data._meta.notifyEmail) || NOTIFY_EMAIL;
    notifyNewHires(before, data, notifyEmail);

    backupSnapshot(doc, before);

    for (const key of TABS) {
      if (!data[key]) continue;
      let sheet = doc.getSheetByName(key);
      if (!sheet) sheet = doc.insertSheet(key);
      sheet.clear();
      const tabData = data[key];

      if (tabData.headers && tabData.headers.length > 0) {
        sheet.appendRow(tabData.headers);
        const headerRange = sheet.getRange(1, 1, 1, tabData.headers.length);
        headerRange.setFontWeight("bold").setBackground("#111827").setFontColor("#ffffff");
        sheet.setFrozenRows(1);

        tabData.headers.forEach((h, i) => {
          if (h.toLowerCase().includes('điện thoại') || h.toLowerCase().includes('phone')) {
            sheet.getRange(2, i + 1, Math.max(tabData.rows.length, 1), 1).setNumberFormat('@');
          }
        });
      }

      if (tabData.rows && tabData.rows.length > 0) {
        const safeRows = tabData.rows.map(row => {
          const newRow = [];
          for (let i = 0; i < tabData.headers.length; i++) newRow.push(row[i] || "");
          return newRow;
        });
        sheet.getRange(2, 1, safeRows.length, tabData.headers.length).setValues(safeRows);
        colorStatusColumn(sheet, tabData.headers, safeRows.length);
      }
      sheet.autoResizeColumns(1, Math.max(tabData.headers.length, 1));
    }

    logActivity(doc, "SYNC_SAVE", `Đồng bộ ${TABS.filter(k => data[k]).length} trang`);
    return ContentService.createTextOutput(JSON.stringify({ status: "success", time: new Date() })).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    logErrorSafe("SYNC_SAVE_ERROR", error);
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.message })).setMimeType(ContentService.MimeType.JSON);
  }
}

// GET dùng để "Tải Từ Cloud". Hỗ trợ ?tab=ungVien để tải nhanh 1 trang riêng lẻ.
function doGet(e) {
  try {
    const doc = SpreadsheetApp.getActiveSpreadsheet();
    const tab = e && e.parameter && e.parameter.tab;
    let result;
    if (tab && TABS.indexOf(tab) > -1) {
      result = {};
      result[tab] = readOneTab(doc, tab);
    } else {
      result = readAllTabs(doc);
    }
    return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    logErrorSafe("LOAD_ERROR", error);
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.message })).setMimeType(ContentService.MimeType.JSON);
  }
}

function readOneTab(doc, key) {
  const sheet = doc.getSheetByName(key);
  if (!sheet || sheet.getLastRow() === 0) return { headers: [], rows: [] };
  const values = sheet.getDataRange().getValues();
  const headers = (values[0] || []).map(h => h.toString());
  const rows = values.slice(1).map(r => r.map(c => (c === null || c === undefined) ? "" : c.toString()));
  return { headers: headers, rows: rows };
}

function readAllTabs(doc) {
  const out = {};
  TABS.forEach(key => { out[key] = readOneTab(doc, key); });
  return out;
}

// Gửi email khi phát hiện dòng nào đó vừa CHUYỂN sang trạng thái "Đã nhận"/"Đạt"
function notifyNewHires(before, after, notifyEmail) {
  if (!notifyEmail) return;
  try {
    const beforeData = before['ungVien'];
    const afterData = after['ungVien'];
    if (!beforeData || !afterData || !afterData.headers) return;

    const statusIdx = afterData.headers.findIndex(h => h.toLowerCase().includes('trạng thái'));
    const nameIdx = afterData.headers.findIndex(h => h.toLowerCase().includes('tên'));
    if (statusIdx === -1) return;

    const wasHired = {};
    (beforeData.rows || []).forEach(r => {
      const key = nameIdx > -1 ? r[nameIdx] : JSON.stringify(r);
      const v = (r[statusIdx] || '').toString().toLowerCase();
      if (v.includes('đã nhận') || v.includes('đạt')) wasHired[key] = true;
    });

    const newlyHired = [];
    (afterData.rows || []).forEach(r => {
      const key = nameIdx > -1 ? r[nameIdx] : JSON.stringify(r);
      const v = (r[statusIdx] || '').toString().toLowerCase();
      const isHired = v.includes('đã nhận') || v.includes('đạt');
      if (isHired && !wasHired[key]) newlyHired.push(r);
    });

    if (newlyHired.length > 0) {
      const names = newlyHired.map(r => nameIdx > -1 ? r[nameIdx] : '(?)').join(', ');
      MailApp.sendEmail({
        to: notifyEmail,
        subject: `[HR IdolRecruit] ${newlyHired.length} ứng viên vừa được nhận`,
        body: `Các ứng viên sau vừa chuyển sang trạng thái "Đã nhận":\n\n${names}\n\nXem chi tiết trong Google Sheet.`
      });
    }
  } catch (err) {
    logErrorSafe("NOTIFY_ERROR", err);
  }
}

// Lưu bản sao lưu (giữ tối đa MAX_BACKUPS bản gần nhất) trước mỗi lần ghi đè
function backupSnapshot(doc, snapshot) {
  try {
    let sheet = doc.getSheetByName('_Backup');
    if (!sheet) {
      sheet = doc.insertSheet('_Backup');
      sheet.appendRow(['Thời gian', 'Dữ liệu (JSON)']);
      sheet.getRange(1, 1, 1, 2).setFontWeight('bold').setBackground('#111827').setFontColor('#ffffff');
      sheet.setFrozenRows(1);
    }
    sheet.insertRowAfter(1);
    sheet.getRange(2, 1, 1, 2).setValues([[new Date(), JSON.stringify(snapshot)]]);

    const lastRow = sheet.getLastRow();
    if (lastRow > MAX_BACKUPS + 1) {
      sheet.deleteRows(MAX_BACKUPS + 2, lastRow - (MAX_BACKUPS + 1));
    }
  } catch (err) {
    logErrorSafe("BACKUP_ERROR", err);
  }
}

// Tô màu cột "Trạng Thái" theo giá trị để nhìn trực quan ngay trên Google Sheets
function colorStatusColumn(sheet, headers, rowCount) {
  const statusIdx = headers.findIndex(h => h.toLowerCase().includes('trạng thái'));
  if (statusIdx === -1 || rowCount === 0) return;
  const colors = {
    'mới': '#dbeafe', 'phỏng vấn': '#fef9c3', 'đã nhận': '#dcfce7',
    'từ chối': '#fee2e2', 'đang đào tạo': '#f3e8ff', 'nghỉ việc': '#fee2e2'
  };
  const range = sheet.getRange(2, statusIdx + 1, rowCount, 1);
  const values = range.getValues();
  const backgrounds = values.map(row => {
    const v = (row[0] || '').toString().toLowerCase();
    for (const key in colors) { if (v.includes(key)) return [colors[key]]; }
    return ['#ffffff'];
  });
  range.setBackgrounds(backgrounds);
}

// Ghi nhật ký mọi lần đồng bộ vào sheet ẩn "_NhatKy"
function logActivity(doc, action, detail) {
  try {
    let sheet = doc.getSheetByName('_NhatKy');
    if (!sheet) {
      sheet = doc.insertSheet('_NhatKy');
      sheet.appendRow(['Thời gian', 'Hành động', 'Chi tiết']);
      sheet.getRange(1, 1, 1, 3).setFontWeight('bold').setBackground('#111827').setFontColor('#ffffff');
      sheet.setFrozenRows(1);
    }
    sheet.appendRow([new Date(), action, detail]);
  } catch (err) { /* im lặng bỏ qua lỗi log */ }
}

function logErrorSafe(action, error) {
  try {
    const doc = SpreadsheetApp.getActiveSpreadsheet();
    logActivity(doc, action, error && error.message ? error.message : String(error));
  } catch (e) { /* bỏ qua */ }
}
