// ==== HR IDOL RECRUIT - APPS SCRIPT BACKEND v4.0 ====
// Khai thác: đa sheet động, backup tự động, nhật ký thay đổi, email thông báo, tải theo tab

const TABS = ['ungVien', 'tienIch', 'noiDung', 'luuTruKhac'];
const NOTIFY_EMAIL = ""; // <-- Điền email HR vào đây để nhận thông báo "Đã nhận"
const MAX_BACKUPS = 10;

function doPost(e) {
  try {
    const doc = SpreadsheetApp.getActiveSpreadsheet();
    const data = JSON.parse(e.postData.contents);

    // 1. Phát hiện ứng viên MỚI đạt trạng thái "Đã nhận" (so với dữ liệu cũ) trước khi ghi đè
    const before = readAllTabs(doc);
    const notifyEmail = (data._meta && data._meta.notifyEmail) || NOTIFY_EMAIL;
    notifyNewHires(before, data, notifyEmail);

    // 2. Sao lưu snapshot hiện tại trước khi ghi đè (an toàn dữ liệu)
    backupSnapshot(doc, before);

    // 3. Ghi dữ liệu mới cho từng sheet (bỏ qua trường _meta không phải là tab dữ liệu)
    for (const key of TABS) {
      if (!data[key]) continue;
      const sheet = doc.getSheetByName(key) || doc.insertSheet(key);
      sheet.clear();
      const tabData = data[key];

      if (tabData.headers && tabData.headers.length > 0) {
        sheet.appendRow(tabData.headers);
        const headerRange = sheet.getRange(1, 1, 1, tabData.headers.length);
        headerRange.setFontWeight("bold").setBackground("#111827").setFontColor("#ffffff");
        sheet.setFrozenRows(1);

        // Định dạng cột SĐT / Điện thoại dạng văn bản để giữ số 0 đầu
        tabData.headers.forEach((h, i) => {
          if (h.toLowerCase().includes('điện thoại') || h.toLowerCase().includes('phone')) {
            sheet.getRange(2, i + 1, Math.max(tabData.rows.length, 1), 1).setNumberFormat('@');
          }
        });
      }

      if (tabData.rows && tabData.rows.length > 0) {
        const safeRows = tabData.rows.map(row => {
          const newRow = [];
          for (let i = 0; i < tabData.headers.length; i++) newRow.push(row[i] || "");
          return newRow;
        });
        sheet.getRange(2, 1, safeRows.length, tabData.headers.length).setValues(safeRows);
        colorStatusColumn(sheet, tabData.headers, safeRows.length);
      }
      sheet.autoResizeColumns(1, Math.max(tabData.headers.length, 1));
    }

    logActivity(doc, "SYNC_SAVE", `Đồng bộ ${TABS.filter(k => data[k]).length} trang, tổng ${countRows(data)} dòng`);
    return ContentService.createTextOutput(JSON.stringify({ status: "success", time: new Date() })).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ error: error.message })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  const doc = SpreadsheetApp.getActiveSpreadsheet();
  const tabParam = e && e.parameter && e.parameter.tab;
  const action = e && e.parameter && e.parameter.action;

  if (action === 'logs') {
    return ContentService.createTextOutput(JSON.stringify(readLogs(doc))).setMimeType(ContentService.MimeType.JSON);
  }
  if (action === 'ping') {
    return ContentService.createTextOutput(JSON.stringify({ status: "ok", time: new Date() })).setMimeType(ContentService.MimeType.JSON);
  }

  if (tabParam) {
    return ContentService.createTextOutput(JSON.stringify(readOneTab(doc, tabParam))).setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(JSON.stringify(readAllTabs(doc))).setMimeType(ContentService.MimeType.JSON);
}

function readOneTab(doc, key) {
  const sheet = doc.getSheetByName(key);
  const result = { headers: [], rows: [] };
  if (sheet && sheet.getLastRow() > 0) {
    const sheetData = sheet.getDataRange().getValues();
    result.headers = sheetData[0].map(String);
    result.rows = sheetData.slice(1).map(r => r.map(String));
  }
  return result;
}

function readAllTabs(doc) {
  const result = {};
  TABS.forEach(key => { result[key] = readOneTab(doc, key); });
  return result;
}

function countRows(data) {
  let n = 0;
  TABS.forEach(k => { if (data[k]) n += (data[k].rows || []).length; });
  return n;
}

// Tô màu nền cột "Trạng thái" ngay trong Google Sheets
function colorStatusColumn(sheet, headers, rowCount) {
  const idx = headers.findIndex(h => h.toLowerCase().includes('trạng thái'));
  if (idx === -1 || rowCount === 0) return;
  const colors = { "đã nhận": "#dcfce7", "phỏng vấn": "#fef9c3", "từ chối": "#fee2e2", "mới": "#dbeafe", "nghỉ việc": "#fee2e2", "đào tạo": "#f3e8ff" };
  const range = sheet.getRange(2, idx + 1, rowCount, 1);
  const values = range.getValues();
  const backgrounds = values.map(v => {
    const s = (v[0] || "").toString().toLowerCase();
    const key = Object.keys(colors).find(k => s.includes(k));
    return [key ? colors[key] : "#ffffff"];
  });
  range.setBackgrounds(backgrounds);
}

// So sánh dữ liệu cũ và mới -> gửi email khi phát hiện ứng viên mới "Đã nhận"
function notifyNewHires(before, incoming, notifyEmail) {
  if (!notifyEmail || !incoming.ungVien) return;
  const oldData = before.ungVien || { headers: [], rows: [] };
  const oldStatusIdx = oldData.headers.findIndex(h => h.toLowerCase().includes('trạng thái'));
  const newHeaders = incoming.ungVien.headers || [];
  const newStatusIdx = newHeaders.findIndex(h => h.toLowerCase().includes('trạng thái'));
  const nameIdx = newHeaders.findIndex(h => h.toLowerCase().includes('tên'));
  if (newStatusIdx === -1) return;

  const oldNames = new Set();
  if (oldStatusIdx > -1) {
    oldData.rows.forEach(r => { if ((r[oldStatusIdx] || "").toLowerCase().includes('đã nhận')) oldNames.add(r[0]); });
  }

  const newHires = incoming.ungVien.rows.filter(r => {
    const isHired = (r[newStatusIdx] || "").toLowerCase().includes('đã nhận');
    const name = nameIdx > -1 ? r[nameIdx] : r[0];
    return isHired && !oldNames.has(name);
  });

  if (newHires.length > 0) {
    const names = newHires.map(r => (nameIdx > -1 ? r[nameIdx] : r[0])).join(", ");
    MailApp.sendEmail(notifyEmail, "🎉 Có ứng viên mới ĐÃ NHẬN!", `Các ứng viên sau vừa được cập nhật trạng thái "Đã nhận":\n\n${names}\n\nXem chi tiết trong Google Sheet.`);
  }
}

// Lưu snapshot dữ liệu trước khi ghi đè, giữ tối đa MAX_BACKUPS bản gần nhất
function backupSnapshot(doc, snapshotData) {
  const sheet = doc.getSheetByName('_Backup') || doc.insertSheet('_Backup');
  if (sheet.getLastRow() === 0) sheet.appendRow(['Thời gian', 'Dữ liệu (JSON)']);
  sheet.appendRow([new Date(), JSON.stringify(snapshotData)]);
  const lastRow = sheet.getLastRow();
  if (lastRow - 1 > MAX_BACKUPS) {
    sheet.deleteRows(2, lastRow - 1 - MAX_BACKUPS);
  }
}

// Ghi nhật ký hoạt động
function logActivity(doc, type, message) {
  const sheet = doc.getSheetByName('_NhatKy') || doc.insertSheet('_NhatKy');
  if (sheet.getLastRow() === 0) sheet.appendRow(['Thời gian', 'Loại', 'Nội dung']);
  sheet.appendRow([new Date(), type, message]);
}

function readLogs(doc) {
  const sheet = doc.getSheetByName('_NhatKy');
  if (!sheet || sheet.getLastRow() < 2) return [];
  return sheet.getRange(2, 1, sheet.getLastRow() - 1, 3).getValues()
    .reverse().slice(0, 50)
    .map(r => ({ time: r[0], type: r[1], message: r[2] }));
}
